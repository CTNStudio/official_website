《中式甲兵》是由星屹工作室制作的中式武器模组，为游戏添加了多种中式盔甲与武器，以及一系列合成材料与新的矿物​。

<!--more-->

## 玩法概述
### 更多盔甲
<img src="https://s41.ax1x.com/2026/07/29/pmfMOwq.png" alt="更多盔甲 - 中式甲兵" style="float:right; width:320px; height:auto;">

加入了29套中式盔甲，其中包括类套不同的铁质中式甲另加其金质、钻石质、下界合金质的变种共28套，以及一套唐代步兵铁甲。

这些盔甲都可以使用游戏中的甲兵锻造台在生存模式合成，也可以通过锻造之石快速升级原版盔甲来获得。

### 更多武器
<img src="https://s41.ax1x.com/2026/07/29/pmfQppF.png" alt="更多武器 - 中式甲兵" style="float:right; width:320px; height:auto;">

加入了3种新的武器类型：戈、偃月刀，和方天画戟。这些武器都具有着独特的技能。

例如戈可以右键将生物钩至身边同时造成伤害，方天画戟在潜行时使用可以对生物造成击退效果并给予负面效果，偃月刀可以潜行右键进行造成持续范围伤害。这些武器也同样可以在生存模式下合成！

### 更多合成材料
<img src="https://s41.ax1x.com/2026/07/29/pmfQATx.png" alt="更多合成材料 - 中式甲兵" style="float:right; width:320px; height:auto;">

本模组还加入了一系列合成材料，提升锻造武器与盔甲的真实感。你可以选择在甲兵锻造台上使用甲片亲手打造盔甲，也可以选择更加便捷的合成方式：使用锻造之石升级原版盔甲。

同时模组还加入了一种新的矿物：远古宝石，用于合成特殊武器。

### 兼容与联动
<img src="https://s41.ax1x.com/2026/07/29/pmfQk01.png" alt="兼容与联动 - 中式甲兵" style="float:right; width:320px; height:auto;">

模组支持使用JEI来查看模组内工作方块的配方，这可以帮助你快速寻找盔甲或武器的合成配方。

如果你安装了[帕秋莉手册](https://www.mcmod.cn/class/1388.html)，我们为你准备了一本指南书，里面包括了模组的所有内容。

<script>
(() => {
  const cfg = document.getElementById('cw-releases');
  if (!cfg) return;
  const repo = cfg.dataset.repo || 'CTNStudio/ChineseWeapons';
  const slots = Array.prototype.slice.call(document.querySelectorAll('.cw-slot'));
  if (!slots.length) return;

  const KEY = 'cw-releases:v3:' + repo;
  const TTL = 30 * 60 * 1000;           // 30 分钟本地缓存

  const esc = (s) => String(s).replace(/[&<>"']/g, c => ({
    '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;'
  }[c]));

  function inline(s) {
    let t = esc(s);
    t = t.replace(/!\[([^\]]*)\]\((https?:\/\/[^)\s]+)\)/g,
      (_, alt, url) => `<img src="${url}" alt="${alt}" loading="lazy">`);
    t = t.replace(/\[([^\]]+)\]\((https?:\/\/[^)\s]+)\)/g,
      (_, text, url) => `<a href="${url}" target="_blank" rel="noopener noreferrer">${text}</a>`);
    t = t.replace(/`([^`\n]+)`/g, '<code>$1</code>');
    t = t.replace(/\*\*([^*\n]+)\*\*/g, '<strong>$1</strong>');
    t = t.replace(/(^|[^*])\*([^*\n]+)\*/g, '$1<em>$2</em>');
    return t;
  }

  function toHtml(md) {
    if (!md) return '';
    const out = [];
    let inList = false;
    const closeList = () => { if (inList) { out.push('</ul>'); inList = false; } };
    const src = String(md).replace(/<br\s*\/?>/gi, '\n').replace(/\r\n?/g, '\n');
    for (const raw of src.split('\n')) {
      const line = raw.replace(/\s+$/, '');
      let m;
      if (!line.trim()) { closeList(); continue; }
      if ((m = line.match(/^\s*#{1,6}\s+(.*)$/))) { closeList(); out.push(`<h4>${inline(m[1])}</h4>`); continue; }
      if (/^\s*(-{3,}|\*{3,}|_{3,})\s*$/.test(line)) { closeList(); out.push('<hr>'); continue; }
      if ((m = line.match(/^\s*>\s?(.*)$/))) { closeList(); out.push(`<blockquote>${inline(m[1])}</blockquote>`); continue; }
      if ((m = line.match(/^\s*[-*+]\s+(.*)$/)) || (m = line.match(/^\s{4,}(\S.*)$/))) {
        if (!inList) { out.push('<ul>'); inList = true; }
        out.push(`<li>${inline(m[1])}</li>`);
        continue;
      }
      closeList();
      out.push(`<p>${inline(line)}</p>`);
    }
    closeList();
    return out.join('');
  }

  function fmtDate(iso) {
    const d = new Date(iso || '');
    if (isNaN(d.getTime())) return '';
    const p = n => String(n).padStart(2, '0');
    return `${d.getFullYear()}-${p(d.getMonth() + 1)}-${p(d.getDate())}`;
  }

  const TAG_RE = /^([A-Za-z]+)-(\d+(?:\.\d+)+)-(.+)$/;
  function parseTag(tag) {
    const m = String(tag || '').trim().match(TAG_RE);
    if (!m) return null;
    return { loader: m[1].toLowerCase(), mc: m[2], version: m[3].trim() };
  }

  const ALL_LOADERS = ['forge', 'neoforge', 'fabric'];
  const NAME = { forge: 'Forge', neoforge: 'NeoForge', fabric: 'Fabric', quilt: 'Quilt' };
  const HEAD_DOT = /^\s*(?:#{1,6}\s*)?[·•]\s*(.+?)\s*$/;
  const HEAD_PLAIN = /^\s*(?:#{1,6}\s*)?(forge|neoforge|fabric)(\s*[,，、\/&]\s*(forge|neoforge|fabric))+\s*$/i;

  function loadersOf(text) {
    return (String(text).match(/\b(forge|neoforge|fabric|quilt)\b/gi) || [])
      .map(x => x.toLowerCase())
      .filter((v, i, a) => a.indexOf(v) === i);
  }

  function parseSections(body) {
    const lines = String(body || '').replace(/<br\s*\/?>/gi, '\n').replace(/\r\n?/g, '\n').split('\n');
    const out = [];
    let cur = { loaders: null, lines: [] };
    const push = () => { if (cur.lines.some(l => l.trim()) || cur.loaders) out.push(cur); };
    for (const line of lines) {
      let ls = null;
      const m = line.match(HEAD_DOT);
      if (m) {
        const found = loadersOf(m[1]);
        if (found.length) ls = found;
      } else if (HEAD_PLAIN.test(line)) {
        ls = loadersOf(line);
      }
      if (ls) { push(); cur = { loaders: ls, lines: [] }; continue; }
      cur.lines.push(line);
    }
    push();
    return out;
  }

  function targets(rel) {
    const p = parseTag(rel.tag_name);
    const secs = parseSections(rel.body);
    const hasExplicit = secs.some(s => s.loaders);
    if (!p) return { all: true };
    if (!hasExplicit) return { all: false, mc: p.mc, loaders: [p.loader] };
    let general = false;
    const ls = [];
    secs.forEach(s => {
      if (!s.loaders) return;
      if (s.loaders.length >= ALL_LOADERS.length) { general = true; return; }
      s.loaders.forEach(l => { if (ls.indexOf(l) < 0) ls.push(l); });
    });
    if (general || !ls.length) return { all: true };
    return { all: false, mc: p.mc, loaders: ls };
  }

  function sectionsFor(rel, loader) {
    return parseSections(rel.body).filter(s =>
      !s.loaders || s.loaders.length >= ALL_LOADERS.length || s.loaders.indexOf(loader) >= 0);
  }

  function labelOf(ls) {
    return ls.map(l => NAME[l] || (l.charAt(0).toUpperCase() + l.slice(1))).join(', ');
  }

  function entry(rel, loader) {
    const p = parseTag(rel.tag_name);
    const url = /^https:\/\//.test(rel.html_url || '') ? esc(rel.html_url) : '';
    const title = esc(p ? p.version : (rel.tag_name || ''));
    const head = url ? `<a href="${url}" target="_blank" rel="noopener noreferrer">${title}</a>` : title;
    const pre = rel.prerelease ? '<span class="cw-badge">预发布</span>' : '';

    let secs = sectionsFor(rel, loader);
    if (!secs.length && String(rel.body || '').trim()) {
      secs = [{ loaders: null, lines: String(rel.body).split('\n') }];
    }

    const parts = [];
    secs.forEach(s => {
      const extra = s.loaders && s.loaders.length < ALL_LOADERS.length;
      if (extra) parts.push(`<h4 class="cw-sub">·${esc(labelOf(s.loaders))}</h4>`);
      const html = toHtml(s.lines.join('\n'));
      if (html) parts.push(html);
    });

    return `<h4 class="cw-head">${head}${pre}<span class="cw-meta">${esc(fmtDate(rel.published_at))}</span></h4>`
      + `<div class="cw-body">${parts.join('') || '<p class="cw-empty">该版本未填写更新说明。</p>'}</div>`;
  }

  function paint(rawList) {
    const list = rawList.map(r => Object.assign({}, r, { parsed: r.parsed || parseTag(r.tag_name) }));
    const known = slots.map(s => ({ loader: s.dataset.loader || '', mc: s.dataset.mc || '' }));
    const info = list.map(r => ({ r: r, t: targets(r) }));

    const orphans = info.filter(x => !x.t.all &&
        !known.some(k => x.t.mc === k.mc && x.t.loaders.indexOf(k.loader) >= 0))
      .map(x => x.r);

    slots.forEach(slot => {
      const loader = slot.dataset.loader || '';
      const mc = slot.dataset.mc || '';
      const hit = info.find(x => x.t.all || (x.t.mc === mc && x.t.loaders.indexOf(loader) >= 0));
      const rel = hit ? hit.r : (orphans[0] || null);
      slot.innerHTML = rel ? entry(rel, loader) : '<p class="cw-empty">暂无该版本的发布记录。</p>';
    });
  }

  try {
    const cached = JSON.parse(localStorage.getItem(KEY) || 'null');
    if (cached && Date.now() - cached.t < TTL && Array.isArray(cached.d) && cached.d.length) {
      paint(cached.d);
      return;
    }
  } catch (e) { /* pass */ }

  fetch(`https://api.github.com/repos/${repo}/releases?per_page=20`)
    .then(r => { if (!r.ok) throw new Error('HTTP ' + r.status); return r.json(); })
    .then(all => {
      const list = (Array.isArray(all) ? all : [])
        .filter(r => r && !r.draft)
        .sort((a, b) => new Date(b.published_at || b.created_at) - new Date(a.published_at || a.created_at))
        .map(r => ({
          tag_name: r.tag_name || '',
          html_url: r.html_url || '',
          published_at: r.published_at || r.created_at,
          body: r.body || '',
          prerelease: !!r.prerelease
        }));
      paint(list);
      try { localStorage.setItem(KEY, JSON.stringify({ t: Date.now(), d: list })); } catch (e) {}
    })
    .catch(err => {
      console.warn('[cw-releases] 读取 GitHub 发布信息失败：', err);
      slots.forEach(s => {
        s.innerHTML = `<p class="cw-empty">暂时无法读取最新发布信息，请前往 <a href="https://github.com/${repo}/releases" target="_blank" rel="noopener noreferrer">GitHub Releases</a> 查看。</p>`;
      });
    });
})();
</script>
