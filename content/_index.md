<script>
(function(){
  var sec=document.querySelector('.sc');
  if(!sec)return;
  var track=sec.querySelector('.sc-track');
  var pin=sec.querySelector('.sc-pin');
  var stage=sec.querySelector('.sc-stage');
  var path=sec.querySelector('.sc-line-bg');
  var cards=[].slice.call(sec.querySelectorAll('.sc-card'));
  var dots=[].slice.call(sec.querySelectorAll('.sc-dot'));
  var n=cards.length;
  if(!n||!path)return;
  var len=0;
  try{len=path.getTotalLength()||0;}catch(e){len=0;}
  var VW=1000,VH=520;
  var STEP=0.19;
  var mq=window.matchMedia('(max-width:900px)');
  var reduce=window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var idx=0,ready=false,introing=false,aborted=false,raf=null;
  var HOLD=1650;
  try{if('scrollRestoration' in history)history.scrollRestoration='manual';}catch(e){}
  window.addEventListener('load',function(){if(!location.hash)window.scrollTo(0,0);});
  try{
    var td=(window.getComputedStyle(path).transitionDuration||'').split(',')[0].trim();
    if(td){
      var tv=parseFloat(td);
      if(!isNaN(tv))HOLD=(/ms$/.test(td)?tv:tv*1000)+60;
    }
  }catch(e){}
  function stageRect(){return stage.getBoundingClientRect();}
  function pointAt(t,r){
    if(!r||!r.width||!r.height)return null;
    var p=path.getPointAtLength(Math.max(0,Math.min(1,t))*len);
    return {x:p.x*(r.width/VW),y:p.y*(r.height/VH)};
  }
  function scaleOf(d){return d===0?1.5:(d===1?0.9:0.74);}
  function alphaOf(d){return d===0?1:(d===1?0.55:0.28);}
  function put(el,t,s,op,z,r){
    var p=pointAt(t,r);
    if(!p){el.style.opacity='0';return;}
    el.style.transform='translate3d('+p.x.toFixed(1)+'px,'+p.y.toFixed(1)+'px,0) translate(-50%,-50%) scale('+s+')';
    el.style.opacity=String(op);
    el.style.zIndex=String(z);
  }
  function render(){
    if(!ready)return;
    var r=stageRect();
    cards.forEach(function(el,i){
      var t=0.5+(idx-i)*STEP;
      var d=Math.abs(i-idx);
      var vis=t>-0.02&&t<1.02;
      el.classList.toggle('is-on',d===0);
      el.style.pointerEvents=vis&&d===0?'auto':'none';
      if(!vis){el.style.opacity='0';el.style.zIndex='0';return;}
      put(el,t,scaleOf(d),alphaOf(d),300-d*10,r);
    });
    dots.forEach(function(d,i){
      d.classList.toggle('is-on',i===idx);
      d.setAttribute('aria-selected',i===idx?'true':'false');
    });
  }
  function progress(){
    var total=track.offsetHeight-pin.offsetHeight;
    if(total<=0)return 0;
    var p=(-track.getBoundingClientRect().top)/total;
    return p<0?0:(p>1?1:p);
  }
  function idxFromProgress(){
    var k=Math.floor(progress()*n);
    return k<0?0:(k>n-1?n-1:k);
  }
  function sync(){
    if(mq.matches)return;
    var next=idxFromProgress();
    if(next!==idx){idx=next;render();}
  }
  function inRange(){
    var r=track.getBoundingClientRect();
    return r.bottom>0&&r.top<window.innerHeight;
  }
  function onScroll(){
    if(introing)abortIntro();
    if(!ready||!inRange())return;
    if(raf)return;
    raf=requestAnimationFrame(function(){raf=null;sync();});
  }
  function goTo(i){
    if(mq.matches)return;
    var total=track.offsetHeight-pin.offsetHeight;
    if(total<=0)return;
    var top=track.getBoundingClientRect().top+window.pageYOffset;
    window.scrollTo({top:Math.round(top+((i+0.5)/n)*total),behavior:reduce?'auto':'smooth'});
  }
  function abortIntro(){
    if(!introing)return;
    aborted=true;
    introing=false;
    ready=true;
    sec.classList.remove('intro');
    cards.forEach(function(c){c.style.transitionDelay='';});
    idx=idxFromProgress();
    render();
  }
  function intro(){
    if(introing||ready)return;
    introing=true;
    sec.classList.add('is-in','intro');
    var start=null,DUR=1100,GAP=150,begun=false;
    function frame(ts){
      if(aborted)return;
      if(!begun){begun=true;start=ts;}
      var el=ts-start-HOLD;
      var done=true;
      var r=stageRect();
      cards.forEach(function(card,i){
        var d=Math.abs(i-idx);
        var target=0.5+(idx-i)*STEP;
        if(el<=i*GAP){card.style.opacity='0';done=false;return;}
        var k=Math.min(1,(el-i*GAP)/DUR);
        if(k<1)done=false;
        var e=1-Math.pow(1-k,3);
        put(card,target*e,scaleOf(d),alphaOf(d)*e,300-d*10,r);
        card.style.pointerEvents='none';
      });
      if(!done){requestAnimationFrame(frame);return;}
      ready=true;
      introing=false;
      sec.classList.remove('intro');
      idx=idxFromProgress();
      render();
    }
    requestAnimationFrame(frame);
  }
  function nearViewport(){
    var r=pin.getBoundingClientRect();
    var h=window.innerHeight||0;
    return r.top<h*1.15&&r.bottom>-h*0.2;
  }
  function autoEnter(){if(nearViewport())intro();}
  if('IntersectionObserver' in window){
    new IntersectionObserver(function(es){
      es.forEach(function(e){if(e.isIntersecting)intro();});
    },{rootMargin:'0px 0px -10% 0px',threshold:0}).observe(pin);
    if(document.readyState==='complete')autoEnter();
    else window.addEventListener('load',autoEnter);
  }else{
    intro();
  }
  cards.forEach(function(card,i){
    card.addEventListener('focus',function(){
      if(ready&&i!==idx)goTo(i);
    });
  });
  dots.forEach(function(d,i){
    d.addEventListener('click',function(){goTo(i);});
  });
  if(reduce){
    ready=true;
    sec.classList.add('is-in');
    idx=idxFromProgress();
    render();
  }else{
    window.addEventListener('scroll',onScroll,{passive:true});
    window.addEventListener('resize',function(){render();sync();},{passive:true});
    if(mq.addEventListener)mq.addEventListener('change',function(){render();sync();});
    else if(mq.addListener)mq.addListener(function(){render();sync();});
  }
})();
</script>
